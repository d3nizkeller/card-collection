        // ============ b364: ОБЩИЙ ПРОГРЕСС МЕЖДУ УСТРОЙСТВАМИ ОДНОГО ПОЛЬЗОВАТЕЛЯ ============
        // Устройствa одного человека (телефон/планшет/ПК) держали баланс и коллекцию
        // КАЖДЫЙ У СЕБЯ (localStorage). Теперь у пользователя есть облачный профиль:
        //   • код профиля (создать на основном устройстве, ввести/открыть ссылку на остальных) —
        //     главная привязка, не зависит от сети;
        //   • авто-привязка по одному IP (b363) — если кода нет, устройства с одним IP
        //     сами попадают в общий профиль (отключается тумблером).
        // Баланс сливается по правилу «последнее состояние + дельты остальных»:
        // устройство пушит не свой баланс, а ИЗМЕНЕНИЕ с момента своей прошлой
        // синхронизации, которое прибавляется к общему значению. Прогресс
        // (паки/карты/коллекция/альбомы/статистика) объединяется, счётчики — максимум.
        const PROF_ALPHABET = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';
        function profHashIp(ip) { // короткий стабильный ключ из IP (сам IP в имя документа не пишем)
            let h = 5381;
            const s = String(ip || '');
            for (let i = 0; i < s.length; i++) h = ((h * 33) ^ s.charCodeAt(i)) >>> 0;
            return h.toString(36);
        }
        function profCode() { try { return String(LS.getItem('nx_prof_code') || '').toUpperCase(); } catch (e) { return ''; } }
        function profIpAuto() { try { return LS.getItem('nx_prof_ipauto') !== '0'; } catch (e) { return true; } }
        function profDocKey() { // ключ облачного документа профиля: код важнее IP
            const c = profCode();
            if (c) return 'c-' + c.toLowerCase();
            if (!profIpAuto()) return '';
            const ip = nxPublicIp();
            return ip ? 'i-' + profHashIp(ip) : '';
        }
        function profPath(key) { return 'https://textdb.dev/api/data/nexus-tcg-prof-' + key; }
        function profNum(v) { const n = Math.floor(Number(v)); return isFinite(n) ? n : 0; }
        function profMaxMap(a, b) {
            const out = {};
            const put = o => { if (!o || typeof o !== 'object') return; for (const k in o) { const v = o[k]; if (typeof v === 'object' && v) { const sub = profMaxMap(out[k], v); out[k] = sub; } else out[k] = Math.max(profNum(out[k]), profNum(v)); } };
            put(a); put(b);
            return out;
        }
        function profMaxStr(a, b) { return String(a || '') >= String(b || '') ? (a || b || '') : (b || ''); }
        function profMergeState(L, R) { // L — локальный слепок, R — общий из облака (или null)
            if (!R) return L;
            const out = {};
            const pm = {};
            (R.packs || []).forEach(p => { if (p && p.id) pm[p.id] = p; });
            (L.packs || []).forEach(p => { if (p && p.id) pm[p.id] = p; });
            out.packs = Object.keys(pm).map(k => pm[k]);
            const cm = {};
            (R.cards || []).forEach(c => { if (c && c.id) cm[c.id] = c; });
            (L.cards || []).forEach(c => { if (c && c.id) cm[c.id] = c; });
            out.cards = Object.keys(cm).map(k => cm[k]);
            out.collection = profMaxMap(R.collection, L.collection);
            out.stats = profMaxMap(R.stats, L.stats);
            out.achievements = profMaxMap(R.achievements, L.achievements);
            out.pity = profMaxMap(R.pity, L.pity);
            out.albumBonus = profMaxMap(R.albumBonus, L.albumBonus);
            out.packStats = profMaxMap(R.packStats, L.packStats);
            out.windowBuys = profMaxMap(R.windowBuys, L.windowBuys);
            out.daily = {
                lastClaimDate: profMaxStr(R.daily && R.daily.lastClaimDate, L.daily && L.daily.lastClaimDate) || null,
                streak: Math.max(profNum(R.daily && R.daily.streak), profNum(L.daily && L.daily.streak)),
                lastDurakWinDate: profMaxStr(R.daily && R.daily.lastDurakWinDate, L.daily && L.daily.lastDurakWinDate) || null
            };
            const md = String(R.missions && R.missions.date || ''), ld = String(L.missions && L.missions.date || '');
            out.missions = (md > ld) ? R.missions : (ld > md ? L.missions : (L.missions && (L.missions.list || []).length >= ((R.missions && R.missions.list) || []).length ? L.missions : R.missions));
            const dk = {};
            (R.deck || []).forEach(id => { dk[id] = 1; });
            (L.deck || []).forEach(id => { dk[id] = 1; });
            out.deck = Object.keys(dk).filter(id => cm[id]);
            return out;
        }
        function profSnap() { // слепок прогресса для облака (без тяжёлых артов: image-ссылки и так внутри паков/карт)
            return {
                coins: Math.max(0, Math.floor(state.coins) || 0),
                packs: (state.packs || []).map(lightPack),
                cards: (state.cards || []).map(lightCard),
                collection: state.collection || {},
                stats: state.stats || {},
                daily: state.daily || {},
                missions: state.missions || {},
                achievements: state.achievements || {},
                pity: state.pity || {},
                deck: state.deck || [],
                albumBonus: state.albumBonus || {},
                packStats: state.packStats || {},
                windowBuys: state.windowBuys || {}
            };
        }
        function profApply(s, mlvl) { // приложить слитое состояние к игре
            if (!s) return;
            state.coins = Math.max(0, profNum(s.coins));
            if (Array.isArray(s.packs)) state.packs = s.packs;
            if (Array.isArray(s.cards)) state.cards = s.cards;
            state.collection = s.collection || {};
            state.stats = Object.assign({}, defaultStats, s.stats || {});
            state.daily = Object.assign({ lastClaimDate: null, streak: 0, lastDurakWinDate: null }, s.daily || {});
            state.missions = s.missions || { date: null, list: [] };
            state.achievements = s.achievements || {};
            state.pity = Object.assign({ packsSinceEpic: 0 }, s.pity || {});
            state.deck = Array.isArray(s.deck) ? s.deck : [];
            state.albumBonus = s.albumBonus || {};
            state.packStats = s.packStats || {};
            state.windowBuys = s.windowBuys || {};
            try { // кирка шахты: уровень не теряется между устройствами
                if (mlvl && typeof miner !== 'undefined' && miner && mlvl > (miner.lvl || 0)) { miner.lvl = mlvl; try { minerSave(); } catch (e) {} }
            } catch (e) {}
            saveState();
            updateCoinDisplay();
            try { refreshVisibleTabs(); } catch (e) {}
        }
        function profMinerLvl() { try { return profNum((typeof miner !== 'undefined' && miner) ? miner.lvl : 0); } catch (e) { return 0; } }
        let profBusy = false, profLastRemote = null, profLastTs = 0, profT = 0;
        function profFetch(key) {
            return fetch(profPath(key) + '?nc=' + Date.now(), { cache: 'no-store' }).then(r => {
                if (!r.ok) throw new Error('HTTP ' + r.status);
                return r.text();
            }).then(txt => { try { return JSON.parse(String(txt || '').trim() || 'null'); } catch (e) { return null; } });
        }
        function profPushDoc(key, doc, keepalive) {
            return fetch(profPath(key), {
                method: 'POST',
                headers: { 'Content-Type': 'text/plain' },
                body: JSON.stringify(doc),
                keepalive: !!keepalive
            }).then(r => { if (!r.ok) throw new Error('HTTP ' + r.status); return true; });
        }
        function profBuildDoc(key, remote, mergedFields, mergedCoins) {
            const did = nxDeviceId();
            const book = Object.assign({}, (remote && remote.book) || {});
            book[did] = { at: Date.now(), bal: mergedCoins, dev: nxDeviceLabel(), ip: nxPublicIp(), n: getPlayerName() };
            return {
                v: 1, ts: Date.now(), by: did, code: profCode(),
                state: Object.assign({}, mergedFields, { coins: mergedCoins }),
                minerLvl: Math.max(profMinerLvl(), profNum(remote && remote.minerLvl)),
                book: book
            };
        }
        function profSyncOnce(force) {
            const key = profDocKey();
            profUI();
            if (!key) return Promise.resolve(false);
            if (profBusy) return Promise.resolve(false);
            profBusy = true;
            const did = nxDeviceId();
            let lastBal = parseInt(LS.getItem('nx_prof_lastbal') || '', 10);
            const first = !isFinite(lastBal);
            if (first) lastBal = 0; // первое подключение: весь мой баланс — это моя дельта
            return profFetch(key).then(remote => {
                profLastRemote = remote;
                const rState = remote && remote.state ? remote.state : null;
                const myDelta = Math.max(0, Math.floor(state.coins) || 0) - lastBal;
                const mergedFields = profMergeState(profSnap(), rState);
                const mergedCoins = Math.max(0, (rState ? profNum(rState.coins) : 0) + myDelta);
                const unionGrew = !rState
                    || (mergedFields.cards || []).length > (rState.cards || []).length
                    || (mergedFields.packs || []).length > (rState.packs || []).length;
                const book = (remote && remote.book) || {};
                const bookStale = !book[did] || (Date.now() - profNum(book[did].at) > 120000);
                const needPush = !!force || !remote || myDelta !== 0 || unionGrew || bookStale;
                // прикладываем общее состояние к себе (чужие карты/паки/баланс приезжают сюда).
                // ВАЖНО: базой «моих изменений» становится именно слитое значение, иначе
                // подтянутый прирост на следующем круге уехал бы в облако второй раз,
                // а трата откатилась бы обратно (тест это ловил: +50 дублировались).
                if (rState) {
                    profApply(Object.assign({}, mergedFields, { coins: mergedCoins }), Math.max(profMinerLvl(), profNum(remote && remote.minerLvl)));
                    try { LS.setItem('nx_prof_lastbal', String(mergedCoins)); } catch (e) {}
                }
                profLastTs = Date.now();
                if (!needPush) { profUI(); return true; }
                const doc = profBuildDoc(key, remote, mergedFields, mergedCoins);
                return profPushDoc(key, doc).then(() => {
                    try { LS.setItem('nx_prof_lastbal', String(mergedCoins)); } catch (e) {}
                    profLastRemote = doc;
                    // защита от перезаписи: читаем ещё раз, если между пушем и чтением
                    // кто-то вписался — повторяем цикл слияния (максимум один раз)
                    return profFetch(key).then(chk => {
                        if (chk && chk.ts && doc.ts && String(chk.ts) !== String(doc.ts) && String(chk.by) !== did) {
                            const r2 = chk.state || null;
                            const d2 = Math.max(0, Math.floor(state.coins) || 0) - mergedCoins;
                            const f2 = profMergeState(profSnap(), r2);
                            const c2 = Math.max(0, (r2 ? profNum(r2.coins) : 0) + d2);
                            profApply(Object.assign({}, f2, { coins: c2 }), Math.max(profMinerLvl(), profNum(chk && chk.minerLvl)));
                            const doc2 = profBuildDoc(key, chk, f2, c2);
                            return profPushDoc(key, doc2).then(() => { try { LS.setItem('nx_prof_lastbal', String(c2)); } catch (e) {} });
                        }
                        return true;
                    });
                }).then(() => { profUI(); return true; });
            }).catch(() => { profUI(); return false; })
              .then(v => { profBusy = false; return v; });
        }
        function profSyncSoon() { clearTimeout(profT); profT = setTimeout(() => { profSyncOnce(false); }, 1500); }
        function profFlush() { // аварийная отправка при скрытии вкладки: моя дельта не должна потеряться
            try {
                const key = profDocKey();
                if (!key) return;
                const lastBal = parseInt(LS.getItem('nx_prof_lastbal') || '', 10);
                const myDelta = Math.max(0, Math.floor(state.coins) || 0) - (isFinite(lastBal) ? lastBal : 0);
                if (myDelta === 0 && profLastRemote) return;
                const rState = profLastRemote && profLastRemote.state ? profLastRemote.state : null;
                const fields = profMergeState(profSnap(), rState);
                const coins = Math.max(0, (rState ? profNum(rState.coins) : 0) + myDelta);
                profPushDoc(key, profBuildDoc(key, profLastRemote, fields, coins), true);
            } catch (e) {}
        }
        function profGenCode() {
            let c = '';
            for (let i = 0; i < 6; i++) c += PROF_ALPHABET[Math.floor(Math.random() * PROF_ALPHABET.length)];
            return c;
        }
        function profSetCode(code, silent) {
            code = String(code || '').toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 12);
            try {
                if (code) LS.setItem('nx_prof_code', code); else LS.setItem('nx_prof_code', '');
                try { LS.removeItem('nx_prof_lastbal'); } catch (e) {} // баланс пересчитаем с нуля против нового профиля
            } catch (e) {}
            profUI();
            if (!silent) profSyncOnce(true);
            return code;
        }
        function profCreateCode() {
            const c = profSetCode(profGenCode());
            try { showToast('Профиль создан: код ' + c + '. Введите этот код (или откройте ссылку ?prof=' + c + ') на остальных устройствах', 'success'); } catch (e) {}
        }
        function profEnterCode() {
            const c = window.prompt('Код профиля с основного устройства (6 знаков) или ссылка с ?prof=…');
            if (!c) return;
            const m = String(c).match(/prof=([a-z0-9-]+)/i);
            const code = profSetCode(m ? m[1] : c);
            if (code) { try { showToast('Код принят: ' + code + ' — синхронизирую…', 'success'); } catch (e) {} }
        }
        function profCopyLink() {
            const c = profCode();
            if (!c) { profCreateCode(); }
            const link = location.origin + location.pathname + '?prof=' + (profCode() || c);
            try { navigator.clipboard && navigator.clipboard.writeText(link); showToast('Ссылка скопирована: ' + link, 'success'); } catch (e) { window.prompt('Скопируйте ссылку для других устройств:', link); }
        }
        function profUnlink() {
            try { LS.setItem('nx_prof_code', ''); LS.removeItem('nx_prof_lastbal'); } catch (e) {}
            profLastRemote = null;
            profUI();
            try { showToast('Привязка снята: устройство снова само по себе', 'info'); } catch (e) {}
        }
        function profToggleIpAuto() {
            try { LS.setItem('nx_prof_ipauto', profIpAuto() ? '0' : '1'); LS.removeItem('nx_prof_lastbal'); } catch (e) {}
            profUI(); profSyncSoon();
        }
        function profLabel() {
            const c = profCode();
            if (c) return 'код ' + c;
            if (!profIpAuto()) return 'выключен';
            const ip = nxPublicIp();
            return ip ? 'авто по IP ' + ip : 'авто по IP (IP ещё не известен)';
        }
        function profUI() {
            const st = document.getElementById('nx-prof-status');
            if (st) {
                const book = (profLastRemote && profLastRemote.book) ? Object.keys(profLastRemote.book).length : 0;
                st.innerHTML =
                    '<span class="text-[10px] text-slate-400">Профиль: <b class="text-slate-200">' + cloudEsc(profLabel()) + '</b></span>' +
                    '<span class="text-[10px] text-slate-500">устройств в профиле: <b class="text-slate-300">' + book + '</b></span>' +
                    '<span class="text-[10px] text-slate-500">синхронизация: <b class="text-slate-300">' + (profLastTs ? statsAgo(profLastTs) : 'ещё не была') + '</b></span>';
            }
            const b = document.getElementById('nx-prof-ipauto-btn');
            if (b) b.innerHTML = '<i class="fa-solid fa-location-dot mr-1"></i>Авто по IP: ' + (profIpAuto() ? 'вкл' : 'выкл');
            const cc = document.getElementById('nx-prof-code-btn');
            if (cc) cc.innerHTML = profCode() ? '<i class="fa-solid fa-key mr-1"></i>Сменить код' : '<i class="fa-solid fa-key mr-1"></i>Создать код';
        }
        function profInit() {
            try {
                const m = String(location.search || '').match(/[?&]prof=([a-zA-Z0-9-]{4,12})/);
                if (m) { profSetCode(m[1].toUpperCase(), true); try { showToast('Открыта ссылка профиля ' + m[1].toUpperCase() + ' — привязываю устройство…', 'success'); } catch (e) {} }
            } catch (e) {}
            profUI();
            setTimeout(() => { profSyncOnce(false); }, 4000);
            setInterval(() => { try { if (!document.hidden) profSyncOnce(false); } catch (e) {} }, 45000);
            document.addEventListener('visibilitychange', () => { try { if (document.hidden) profFlush(); else profSyncSoon(); } catch (e) {} });
            window.addEventListener('pagehide', () => { try { profFlush(); } catch (e) {} });
        }
